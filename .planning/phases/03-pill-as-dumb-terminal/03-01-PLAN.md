# Plan 03-01: Refactor pill to stateless renderer

<objective>
Remove all state logic from pill.html so it becomes a pure render target. Main process sends
`render` payloads describing what the pill should look like; pill renders exactly that and
forwards clicks back to main. Zero local state variables, zero timers, zero decision-making
in the pill. This eliminates the pill as an independent state tracker (currently one of three)
and makes main process the sole source of truth for what the pill displays.
</objective>

Replace the entire preload-pill.js API with a minimal dumb-terminal surface.

**Remove these exposed methods:**
- `onStateUpdate` (replaced by `onRender`)
- `onLevelUpdate` (level is now embedded in `render` payload)
- `toggleRecording` (pill no longer decides what toggle means)
- `cancelRecording` (pill no longer decides what cancel means)
- `onShowEnterButton` (enter-ready is now just a `shape` in the render payload)
- `simulateEnter` (pill just sends `pill-clicked` — main decides)
- `getState` (pill doesn't poll state — it renders what it's told)

**Add these exposed methods:**
- `onRender(cb)` — receives `{ shape, level, autoEnterMode, message }` from main via `ipcRenderer.on('pill-render', ...)`
- `click(action)` — sends `ipcRenderer.invoke('pill-clicked', action)` where `action` is one of `'capsule'`, `'stop'`, `'cancel'`, `'enter'`

**Keep these unchanged:**
- `setIgnoreMouse(ignore)` — click-through toggle, purely a window concern
- `showContextMenu()` — right-click menu, purely a window concern

**Note:** The `get-state` IPC handler in main.js is NOT removed. It is still used by the main window (`preload.js`) and tests. Only the pill's `getState` preload method is removed.

Resulting preload-pill.js:
```js
const { contextBridge, ipcRenderer } = require('electron');

contextBridge.exposeInMainWorld('electronAPI', {
  // Render — main process tells pill exactly what to display
  onRender: (cb) => ipcRenderer.on('pill-render', (_, data) => cb(data)),

  // Click — pill tells main what was clicked, main decides the action
  click: (action) => ipcRenderer.invoke('pill-clicked', action),

  // Window-level concerns (unchanged)
  setIgnoreMouse: (ignore) => ipcRenderer.send('pill-set-ignore-mouse', ignore),
  showContextMenu: () => ipcRenderer.invoke('pill-context-menu'),
});
```
</action>

  capsule.className = 'capsule ' + shape;

  // Tooltip: error → show error text; dormant → show default; all else → hide
  if (shape === 'error' && message) {
    tooltip.innerHTML = message;
    tooltip.classList.add('tooltip-error');
    tooltip.style.display = '';  // errors always show tooltip
  } else if (shape === 'dormant') {
    tooltip.innerHTML = defaultTooltipHTML;
    tooltip.classList.remove('tooltip-error');
    tooltip.style.display = '';  // dormant shows on hover (CSS-driven)
  } else {
    tooltip.classList.remove('tooltip-error');
    tooltip.style.display = 'none';  // force-hide — CSS :hover can linger after click
  }

  // Processing: stagger bar animation delays
  if (shape === 'processing') {
    for (let i = 0; i < NUM_BARS; i++) {
      barEls[i].style.height = '6px';
      barEls[i].style.animationDelay = (i * 0.06) + 's';
    }

  // Stop button icon: show enter icon when autoEnterMode === 'auto'
  const showEnter = autoEnterMode === 'auto';
  stopSquare.style.display = showEnter ? 'none' : '';
  stopEnterIcon.style.display = showEnter ? '' : 'none';
}

/**
 * Send a render payload to the pill. This is the ONLY way main talks to the pill
 * about visual state. The pill has no state of its own — it renders exactly this.

function renderPill(overrides = {}) {
  if (!pillWindow || pillWindow.isDestroyed()) return;
  const settings = store ? store.getSettings() : {};
  const payload = {
    shape: sm.state,           // dormant | recording | processing | success | error
    level: 0,                  // audio level — only meaningful during recording
    autoEnterMode: settings.autoEnterMode || 'off',
    message: sm.message || '', // error message text — only meaningful for shape='error'
    ...overrides,              // caller can override shape (e.g., 'enter-ready') or level
  };
  pillWindow.webContents.send('pill-render', payload);
}

```js
ipcMain.handle('pill-clicked', async (_, action) => {
  log.ipc('pill-clicked', `action=${action} state=${sm.state}`);

  if (action === 'enter') {
    // Enter button clicked — simulate Enter keystroke, then go dormant
    simulateEnter();
    setAppState('dormant');
    broadcastState();
    return;
  }

  if (action === 'cancel') {
    const gate = canAcceptAction('cancel');
    if (!gate.allowed) return { success: false, error: gate.error };
    // Delegate to same logic as cancel-processing IPC handler:
    // 1. Reset state immediately so UI unblocks
    setAppState('dormant');
    broadcastState();
    // 2. Await sidecar cancel — must match cancel-processing behavior exactly
    if (!sidecar || !sidecar.isRunning) return { success: true };
    try {
      await sidecar.send('cancel');
      return { success: true };
    } catch (err) {
      return { success: false, error: err.message };
    }
  }

  if (action === 'stop') {
    const gate = canAcceptAction('stop');
    if (!gate.allowed) return;
    await toggleRecording();
    return;
  }

  // action === 'capsule' — dormant click or enter-ready click
  // For enter-ready: main knows the current shape it sent, treat as enter
  // Note: success/error clicks fall through — canAcceptAction('toggle') handles them

  // Capture foreground before starting — pill is non-focusable so target app is still fg
  if (sidecar && sidecar.isRunning) sidecar.send('capture_fg').catch(() => {});

  const gate = canAcceptAction('toggle');
  if (!gate.allowed) {
    broadcastError(gate.error);
    return;
  }

  if (gate.actualAction === 'stop' || gate.actualAction === 'cancel') {
    await toggleRecording();
  } else if (mainWindow && !mainWindow.isDestroyed()) {
    mainWindow.webContents.executeJavaScript('triggerTrustedHotkeyToggle()').catch(() => {});
  } else {
    toggleRecording();
  }
});
```
